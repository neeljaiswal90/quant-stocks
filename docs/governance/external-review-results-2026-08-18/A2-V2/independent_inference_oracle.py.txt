class OracleError(Exception):
    def __init__(self, reason: str, detail: str) -> None:
        self.reason = reason
        self.detail = detail
        super().__init__(f"{reason}: {detail}")


def _context() -> Context:
    context = Context(
        prec=DECIMAL_PRECISION,
        rounding=ROUND_HALF_EVEN,
        Emin=-999_999,
        Emax=999_999,
        capitals=1,
        clamp=0,
    )
    for quiet in (Clamped, Underflow, Subnormal, Inexact, Rounded):
        context.traps[quiet] = False
    for trapped in (InvalidOperation, DivisionByZero, Overflow):
        context.traps[trapped] = True
    context.clear_flags()
    return context


def _plain(value: Decimal) -> str:
    with localcontext(_context()):
        return format(value.quantize(_QUANT), "f")

def is_canonical_decimal(value: object) -> tuple[bool, str]:
    """Pure-string canonical-decimal predicate. Never constructs Decimal."""

    if type(value) is not str:
        return False, "not a str"
    if len(value) > _MAX_CANONICAL_DECIMAL_LENGTH:
        return False, "length exceeds 128"
    if _INDEPENDENT_DECIMAL_PATTERN.fullmatch(value) is None:
        return False, "grammar fullmatch failed"
    if value.startswith("-") and not any(ch in _NONZERO_DIGITS for ch in value):
        return False, "negative zero"
    return True, "ok"


def require_canonical_decimal(value: object, *, field: str) -> str:
    ok, why = is_canonical_decimal(value)
    if not ok:
        raise OracleError("NO_GO_FAIL_CLOSED", f"{field}: {why}")
    return value  # type: ignore[return-value]


def require_canonical_series(values: object, *, kind: str) -> list[str]:
    if type(values) is not list or not values:
        raise OracleError("NO_GO_FAIL_CLOSED", f"{kind} must be a non-empty list")
    out: list[str] = []
    for i, value in enumerate(values):
        out.append(require_canonical_decimal(value, field=f"{kind}[{i}]"))
    return out

def _cube_root(value: Decimal) -> Decimal:
    """Newton cube-root at registered precision. Independent implementation."""

    if value <= 0 or not value.is_finite():
        raise OracleError("NO_GO_NO_FALLBACK", "cube-root radicand must be finite and positive")
    with localcontext(_context()) as context:
        # Extra guard digits during iteration, then fold back to registered prec.
        context.prec = DECIMAL_PRECISION + 16
        # Decimal.adjusted() is the exponent of the leading digit (scientific).
        estimate = Decimal(10) ** (value.adjusted() // 3)
        two, three = Decimal(2), Decimal(3)
        for _ in range(256):
            updated = (two * estimate + value / (estimate * estimate)) / three
            scale = max(updated.adjusted(), estimate.adjusted()) - context.prec + 2
            tolerance = Decimal(1).scaleb(scale)
            if updated == estimate or abs(updated - estimate) <= tolerance:
                estimate = updated
                break
            estimate = updated
        else:
            raise OracleError("NO_GO_NO_FALLBACK", "cube-root iteration did not converge")
        context.prec = DECIMAL_PRECISION
        return +estimate


def select_block_length(deltas: list[Decimal]) -> tuple[str, int, int]:
    """Corrected Politis-White automatic selector, then registered bounds.

    Returns (raw_block_length_plain, common_block_length, selected_lag).
    """

    count = len(deltas)
    if count < MIN_SERIES_LENGTH:
        raise OracleError(
            "NO_GO_NO_FALLBACK",
            f"series length {count} < registered minimum {MIN_SERIES_LENGTH}",
        )
    with localcontext(_context()):
        n = Decimal(count)
        mean = sum(deltas, Decimal(0)) / n
        centered = [value - mean for value in deltas]
        k_n = max(5, int((n.log10().sqrt()).to_integral_value(rounding=ROUND_CEILING)))
        m_max = int(n.sqrt().to_integral_value(rounding=ROUND_CEILING)) + k_n
        autocovariances = [
            sum((centered[i] * centered[i + lag] for i in range(count - lag)), Decimal(0)) / n
            for lag in range(m_max + 1)
        ]
        gamma_zero = autocovariances[0]
        if gamma_zero.is_zero():
            raise OracleError("NO_GO_NO_FALLBACK", "gamma_hat(0) is exactly zero")
        threshold = Decimal(2) * (n.log10() / n).sqrt()
        autocorrelations = [value / gamma_zero for value in autocovariances]
        selected_lag: int | None = None
        for candidate in range(1, m_max - k_n + 2):
            if all(
                abs(autocorrelations[lag]) < threshold
                for lag in range(candidate, candidate + k_n)
            ):
                selected_lag = candidate
                break
        if selected_lag is None:
            raise OracleError(
                "NO_GO_NO_FALLBACK",
                "no full K_N autocorrelation window is strictly insignificant",
            )
        lag_window = min(2 * selected_lag, m_max)
        lag_window_decimal = Decimal(lag_window)

        def flat_top(lag: int) -> Decimal:
            ratio = Decimal(lag) / lag_window_decimal
            if Decimal(2) * ratio <= 1:
                return Decimal(1)
            return Decimal(2) * (Decimal(1) - ratio)

        weighted_lag_moment = Decimal(2) * sum(
            (flat_top(lag) * Decimal(lag) * autocovariances[lag] for lag in range(1, lag_window + 1)),
            Decimal(0),
        )
        spectral_zero = gamma_zero + Decimal(2) * sum(
            (flat_top(lag) * autocovariances[lag] for lag in range(1, lag_window + 1)),
            Decimal(0),
        )
        denominator = Decimal(2) * spectral_zero * spectral_zero
        if not denominator.is_finite() or denominator.is_zero():
            raise OracleError("NO_GO_NO_FALLBACK", "D_hat_SB is zero, nonfinite, or undefined")
        radicand = Decimal(2) * weighted_lag_moment * weighted_lag_moment * n / denominator
        raw = _cube_root(radicand)
        integerized = int(raw.to_integral_value(rounding=ROUND_CEILING))
        floor_cap = int((n / Decimal(4)).to_integral_value(rounding=ROUND_FLOOR))
        common = min(floor_cap, max(MIN_BLOCK_LENGTH, integerized))
        if common < 1:
            raise OracleError("NO_GO_NO_FALLBACK", "bounded common block length is nonpositive")
        return _plain(raw), common, selected_lag


def stationary_bootstrap_indices(
    *,
    series_length: int,
    replicate_count: int,
    mean_block_length: int,
    seed: int,
) -> tuple[tuple[int, ...], ...]:
    """Independent Politis-Romano index assembly on the registered PCG32 stream.

    Each block starts at an independently uniform series index, has an exact
    integer-geometric length, wraps around the series, and a final block is
    truncated so every replicate has exactly ``series_length`` indices. The
    RNG stream is shared across replicates in output order.
    """

    generator = Pcg32.from_seed(seed)
    output: list[tuple[int, ...]] = []
    for _ in range(replicate_count):
        replicate: list[int] = []
        while len(replicate) < series_length:
            start = generator.bounded_index(series_length)
            block_length = generator.geometric(mean_block_length)
            remaining = series_length - len(replicate)
            take = min(block_length, remaining)
            replicate.extend((start + offset) % series_length for offset in range(take))
        output.append(tuple(replicate))
    return tuple(output)


def annualized_mean(values: list[Decimal]) -> Decimal:
    n = Decimal(len(values))
    return Decimal(ANNUALIZATION) * (sum(values, Decimal(0)) / n)


def bootstrap_statistics(
    deltas: list[Decimal], indices: tuple[tuple[int, ...], ...]
) -> list[Decimal]:
    with localcontext(_context()):
        annualization = Decimal(ANNUALIZATION)
        n = Decimal(len(deltas))
        stats: list[Decimal] = []
        for replicate in indices:
            total = sum((deltas[i] for i in replicate), Decimal(0))
            stats.append(annualization * (total / n))
        return stats


def canonical_hash(values: list[Decimal]) -> str:
    payload = json.dumps([_plain(v) for v in values], separators=(",", ":")).encode("utf-8")
    digest = hashlib.sha256(payload).hexdigest()
    return ":".join(digest[i : i + 8] for i in range(0, 64, 8))


def newey_west_diagnostic(deltas: list[Decimal]) -> tuple[int, str, str, str]:
    count = len(deltas)
    if count < 2:
        raise OracleError("DIAGNOSTIC_UNAVAILABLE_NO_PRIMARY_FALLBACK", "n < 2")
    with localcontext(_context()):
        n = Decimal(count)
        ratio = n / Decimal(100)
        two_ninths = Decimal(2) / Decimal(9)
        power = (two_ninths * ratio.ln()).exp()
        q = min(count - 1, int((Decimal(4) * power).to_integral_value(rounding=ROUND_FLOOR)))
        if q < 0:
            raise OracleError("DIAGNOSTIC_UNAVAILABLE_NO_PRIMARY_FALLBACK", "lag q is negative")
        mean = sum(deltas, Decimal(0)) / n
        centered = [value - mean for value in deltas]

        def gamma(k: int) -> Decimal:
            return sum(
                (centered[t] * centered[t - k] for t in range(k, count)), Decimal(0)
            ) / n

        gamma_zero = gamma(0)
        omega = gamma_zero + Decimal(2) * sum(
            (
                (Decimal(1) - Decimal(k) / Decimal(q + 1)) * gamma(k)
                for k in range(1, q + 1)
            ),
            Decimal(0),
        )
        if not omega.is_finite() or omega < 0:
            raise OracleError(
                "DIAGNOSTIC_UNAVAILABLE_NO_PRIMARY_FALLBACK",
                "Omega_hat is negative or nonfinite",
            )
        se_annualized = Decimal(ANNUALIZATION) * (omega / n).sqrt()
        annualized = Decimal(ANNUALIZATION) * mean
        if se_annualized.is_zero():
            raise OracleError(
                "DIAGNOSTIC_UNAVAILABLE_NO_PRIMARY_FALLBACK",
                "annualized standard error is zero",
            )
        t_stat = annualized / se_annualized
        return q, _plain(omega), _plain(se_annualized), _plain(t_stat)


@dataclass(frozen=True)
class IndependentHolm:
    family_size: int
    ordered_p_values: tuple[str, ...]
    adjusted_thresholds: tuple[str, ...]
    rejected: tuple[bool, ...]


def holm_step_down_independent(p_values: object, *, alpha: object = "0.05") -> IndependentHolm:
    series = require_canonical_series(p_values, kind="holm_p")
    checked_alpha = require_canonical_decimal(alpha, field="alpha")
    with localcontext(_context()):
        alpha_d = Decimal(checked_alpha)
        if alpha_d <= 0 or alpha_d >= 1:
            raise OracleError(
                "NO_GO_FAIL_CLOSED",
                "alpha must be a significance level in the open interval (0, 1)",
            )
        parsed: list[Decimal] = []
        for i, value in enumerate(series):
            p = Decimal(value)
            if not (Decimal(0) <= p <= Decimal(1)):
                raise OracleError("NO_GO_FAIL_CLOSED", f"holm_p[{i}] outside [0, 1]")
            parsed.append(p)
        m = len(parsed)
        order = sorted(range(m), key=lambda i: parsed[i])
        thresholds: list[Decimal] = []
        rejected_sorted: list[bool] = []
        still = True
        for rank, _ in enumerate(order):
            thr = alpha_d / Decimal(m - rank)
            thresholds.append(thr)
            reject = still and parsed[order[rank]] <= thr
            if not reject:
                still = False
            rejected_sorted.append(reject)
        return IndependentHolm(
            m,
            tuple(_plain(parsed[i]) for i in order),
            tuple(_plain(t) for t in thresholds),
            tuple(rejected_sorted),
        )


@dataclass(frozen=True)
class IndependentInference:
    series_length: int
    point_estimate: str
    raw_block_length: str
    common_block_length: int
    selected_lag: int
    one_sided_95_lcb: str
    two_sided_90_low: str
    two_sided_90_high: str
    bootstrap_distribution_sha256: str
    replicate_count: int
    seed: int
    nw_lag: int
    nw_long_run_variance: str
    nw_annualized_standard_error: str
    nw_t_statistic: str


def run_inference_independent(deltas: object) -> IndependentInference:
    strings = require_canonical_series(deltas, kind="delta")
    with localcontext(_context()):
        series = [Decimal(s) for s in strings]
    raw, common, lag = select_block_length(series)
    indices = stationary_bootstrap_indices(
        series_length=len(series),
        replicate_count=BOOTSTRAP_REPLICATES,
        mean_block_length=common,
        seed=BOOTSTRAP_SEED,
    )
    stats = bootstrap_statistics(series, indices)
    with localcontext(_context()):
        point = annualized_mean(series)
        ordered = sorted(stats)
        lcb = ordered[ONE_SIDED_95_LCB_RANK - 1]
        low = ordered[TWO_SIDED_90_RANKS[0] - 1]
        high = ordered[TWO_SIDED_90_RANKS[1] - 1]
    q, omega, se, t = newey_west_diagnostic(series)
    return IndependentInference(
        series_length=len(series),
        point_estimate=_plain(point),
        raw_block_length=raw,
        common_block_length=common,
        selected_lag=lag,
        one_sided_95_lcb=_plain(lcb),
        two_sided_90_low=_plain(low),
        two_sided_90_high=_plain(high),
        bootstrap_distribution_sha256=canonical_hash(stats),
        replicate_count=BOOTSTRAP_REPLICATES,
        seed=BOOTSTRAP_SEED,
        nw_lag=q,
        nw_long_run_variance=omega,
        nw_annualized_standard_error=se,
        nw_t_statistic=t,
    )

    out: list[str] = []
    x = seed
    for _ in range(n):
        x = (x * 1103515245 + 12345) % (2**31)
        out.append(f"{Decimal(x % 2001 - 1000) / Decimal(100000) + Decimal('0.0012'):.6f}")
    return out


def _grouped_sha256(data: bytes) -> str:
    h = hashlib.sha256(data).hexdigest()
    return ":".join(h[i : i + 8] for i in range(0, 64, 8))

    for good in _ALPHA_IN:

    # 4. KAT fixture: independent recomputation vs frozen expected
