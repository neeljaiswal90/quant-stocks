def grouped_sha256(path: Path) -> str:
    digest = hashlib.sha256(path.read_bytes()).hexdigest()
    return ":".join(digest[i : i + 8] for i in range(0, 64, 8))


def check(label: str, expected: dict[str, str]) -> list[str]:
    failures: list[str] = []
    print(f"== {label} ==")
    for rel, exp in expected.items():
        path = ROOT / rel
        if not path.is_file():
            print(f"MISSING  {rel}")
            failures.append(f"missing {rel}")
            continue
        got = grouped_sha256(path)
        status = "MATCH" if got == exp else "MISMATCH"
        print(f"{status}  {got}  {path.stat().st_size:8d}  {rel}")
        if got != exp:
            print(f"  expected {exp}")
            failures.append(f"mismatch {rel}")
    return failures


def main() -> int:
    failures = check("packet-bound files", PACKET_EXPECTED)
    vector = ROOT / (
        "tests/fixtures/governance/xnas-ordered-session-vector-2010-2027-v1.candidate.json"
    )
    vector_hash = grouped_sha256(vector)
    print("== ordered session vector (recorded inside evidence JSON) ==")
    print(f"OBSERVED  {vector_hash}  {vector.stat().st_size:8d}  {vector.relative_to(ROOT)}")
    evidence = json.loads(
        (ROOT / "configs/governance/xnas-session-calendar-evidence-v1.json").read_text(
            encoding="utf-8"
        )
    )
    bound_vector = evidence["artifacts"]["ordered_session_vector"]["sha256"]
    if vector_hash != bound_vector:
        print(f"MISMATCH evidence artifacts.ordered_session_vector.sha256={bound_vector}")
        failures.append("vector hash vs evidence")
    else:
        print("MATCH    evidence artifacts.ordered_session_vector.sha256")

    print("== hashes.json leaf re-check ==")
    manifest = json.loads(
        (ROOT / "configs/governance/xnas-session-calendar-evidence-v1.hashes.json").read_text(
            encoding="utf-8"
        )
    )
    for row in manifest["artifacts"]:
        rel = row["path"]
        got = grouped_sha256(ROOT / rel)
        status = "MATCH" if got == row["sha256"] else "MISMATCH"
        print(f"{status}  {rel}")
        if got != row["sha256"]:
            failures.append(f"manifest mismatch {rel}")

    failures.extend(check("registered-authority files cited by evidence JSON", AUTHORITY_EXPECTED))

    print()
    if failures:
        print(f"RESULT FAIL ({len(failures)})")
        for item in failures:
            print(f"  - {item}")
        return 1
    print("RESULT PASS all packet-bound, evidence-bound, manifest, and authority hashes match")
    return 0


if __name__ == "__main__":
