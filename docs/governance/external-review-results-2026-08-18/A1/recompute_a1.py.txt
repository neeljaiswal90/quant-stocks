def grouped_sha256(data: bytes) -> str:
    digest = hashlib.sha256(data).hexdigest()
    return ":".join(digest[i : i + 8] for i in range(0, 64, 8))


def file_grouped(path: Path) -> str:
    return grouped_sha256(path.read_bytes())


def independent_canonical_json_bytes(document: Any) -> bytes:
    """Independently written canonical JSON: sorted keys, compact, UTF-8, trailing NL."""
    return (
        json.dumps(
            document,
            ensure_ascii=False,
            allow_nan=False,
            sort_keys=True,
            separators=(",", ":"),
        ).encode("utf-8")
        + b"\n"
    )


def independent_semantic_sha256(document: dict[str, Any]) -> str:
    semantic = dict(document)
    semantic.pop("semantic_sha256", None)
    return grouped_sha256(independent_canonical_json_bytes(semantic))


def check(condition: bool, label: str, detail: str = "") -> None:
    if condition:
        PASSES.append(label)
        extra = f" — {detail}" if detail else ""
        print(f"PASS  {label}{extra}")
    else:
        FAILURES.append(label)
        extra = f" — {detail}" if detail else ""
        print(f"FAIL  {label}{extra}")


def load_json(path: Path) -> dict[str, Any]:
    def reject_duplicates(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
        out: dict[str, Any] = {}
        for key, value in pairs:
            if key in out:
                raise ValueError(f"duplicate JSON key: {key}")
            out[key] = value
        return out

    value = json.loads(path.read_text("utf-8"), object_pairs_hook=reject_duplicates)
    if not isinstance(value, dict):
        raise TypeError(f"{path} is not a JSON object")
    return value

def verify_lineage_hashes(config: dict[str, Any]) -> None:
    print("\n=== 4. Independent lineage predecessor hashes ===")
    lineage = config["lineage"]
    check(
        lineage["protected_main_at_authoring"] == EXPECTED_PROTECTED_MAIN,
        "protected_main_at_authoring matches parent commit grouped prefix",
    )
    parent = git(["rev-parse", "HEAD^"])
    reconstructed = ":".join(parent[i : i + 8] for i in range(0, 40, 8))
    check(
        reconstructed == EXPECTED_PROTECTED_MAIN,
        "parent commit SHA reconstructs protected_main_at_authoring",
        f"{parent} -> {reconstructed}",
    )
    checked = 0
    for name, binding in lineage.items():
        if not isinstance(binding, dict) or "path" not in binding or "sha256" not in binding:
            continue
        rel = binding["path"]
        observed = file_grouped(WORKTREE / rel)
        check(observed == binding["sha256"], f"lineage {name} {rel}", observed)
        checked += 1
    check(checked == 9, "exactly nine path-bound lineage predecessors", str(checked))
    serialized = json.dumps(config)
    check(
        PR26_PATH not in serialized,
        "A1 record does not mention PR #26 owner-selection path",
    )
    check(
        "ppw_bootstrap_owner_selections" not in lineage,
        "A1 lineage object has no ppw_bootstrap_owner_selections key",
        str(list(lineage.keys())),
    )

def extract_yaml_block(doc_text: str) -> str:
    match = re.search(r"```yaml\n(.*?)```", doc_text, re.S)
    if not match:
        raise RuntimeError("canonical YAML block missing from bound doc")
    return match.group(1)

def independent_claims_ok(document: dict[str, Any]) -> tuple[bool, str]:
    claims = document.get("claims")
    if not isinstance(claims, dict):
        return False, "claims missing"
    if any(claims.get(field) is not False for field in FORBIDDEN_TRUE):
        return False, "forbidden claim not False"
    if any(claims.get(field) is not True for field in REQUIRED_TRUE):
        return False, "required claim not True"
    if claims.get("registration_meaning") != REGISTRATION_MEANING:
        return False, "registration meaning changed"
    return True, "ok"


def verify_tamper(config: dict[str, Any]) -> None:
    print("\n=== 8. Tamper rejection on copies OUTSIDE the worktree ===")
    if TAMPER_DIR.exists():
        shutil.rmtree(TAMPER_DIR)
    TAMPER_DIR.mkdir(parents=True)

    config_src = WORKTREE / "configs/governance/owner-decision-record-2026-08-16-v1.json"
    schema_src = WORKTREE / "schemas/governance/owner-decision-record-2026-08-16-v1.schema.json"
    manifest_src = WORKTREE / "configs/governance/owner-decision-record-2026-08-16-v1.hashes.json"
    lineage_src = WORKTREE / "qme/quant/capacity_solver.py"

    # 8a. Byte flip of config copy — hash must diverge from pin.
    mutated_config = TAMPER_DIR / "mutated-config.json"
    raw = bytearray(config_src.read_bytes())
    raw[0] ^= 0x01
    mutated_config.write_bytes(bytes(raw))
    check(
        file_grouped(mutated_config) != BOUND[config_src.relative_to(WORKTREE).as_posix()],
        "tamper: flipped config byte changes SHA-256 vs pin",
        file_grouped(mutated_config),
    )
    check(
        file_grouped(config_src) == BOUND[config_src.relative_to(WORKTREE).as_posix()],
        "tamper: original worktree config bytes unchanged after copy mutation",
    )

    # 8b. Semantic-hash field rewrite.
    forged = json.loads(config_src.read_text("utf-8"))
    forged["semantic_sha256"] = "00000000:00000000:00000000:00000000:00000000:00000000:00000000:00000000"
    semantic_copy = TAMPER_DIR / "forged-semantic.json"
    semantic_copy.write_text(json.dumps(forged, indent=2) + "\n", "utf-8")
    recomputed = independent_semantic_sha256(forged)
    check(
        recomputed == EXPECTED_SEMANTIC,
        "tamper: rewriting only the semantic_sha256 field leaves true digest unchanged",
        recomputed,
    )
    check(
        forged["semantic_sha256"] != recomputed,
        "tamper: forged semantic_sha256 field no longer equals independent digest",
    )

    # 8c. Mutate each forbidden claim to True in a copy.
    for key in FORBIDDEN_TRUE:
        forged_claims = json.loads(config_src.read_text("utf-8"))
        forged_claims["claims"][key] = True
        ok, reason = independent_claims_ok(forged_claims)
        check(not ok, f"tamper: setting {key}=True fails claims contract", reason)
        path = TAMPER_DIR / f"claim-true-{key}.json"
        path.write_text(json.dumps(forged_claims), "utf-8")
        check(
            file_grouped(path) != BOUND[config_src.relative_to(WORKTREE).as_posix()],
            f"tamper: {key}=True copy hash diverges from pin",
        )

    # 8d. Mutate each required claim to False.
    for key in REQUIRED_TRUE:
        forged_claims = json.loads(config_src.read_text("utf-8"))
        forged_claims["claims"][key] = False
        ok, reason = independent_claims_ok(forged_claims)
        check(not ok, f"tamper: setting {key}=False fails claims contract", reason)

    # 8e. Mutate registration meaning.
    forged_meaning = json.loads(config_src.read_text("utf-8"))
    forged_meaning["claims"]["registration_meaning"] = "DECISIONS_REGISTERED_AND_BLOCKERS_CLEARED"
    ok, reason = independent_claims_ok(forged_meaning)
    check(not ok, "tamper: forged registration_meaning fails claims contract", reason)

    # 8f. Mutate a lineage predecessor file copy.
    lineage_copy = TAMPER_DIR / "mutated-capacity-solver.py"
    lineage_bytes = bytearray(lineage_src.read_bytes())
    lineage_bytes[-1] ^= 0x01
    lineage_copy.write_bytes(bytes(lineage_bytes))
    expected_lineage = config["lineage"]["capacity_solver_code"]["sha256"]
    check(
        file_grouped(lineage_copy) != expected_lineage,
        "tamper: mutated predecessor file hash diverges from A1 lineage pin",
        file_grouped(lineage_copy),
    )
    check(
        file_grouped(lineage_src) == expected_lineage,
        "tamper: original predecessor file in worktree unchanged",
    )

    # 8g. Mutate schema const pin copy.
    schema_forged = json.loads(schema_src.read_text("utf-8"))
    schema_forged["const"]["claims"]["production_ready"] = True
    schema_copy = TAMPER_DIR / "mutated-schema.json"
    schema_copy.write_text(json.dumps(schema_forged), "utf-8")
    original_schema = json.loads(schema_src.read_text("utf-8"))
    check(
        schema_forged["const"] != original_schema["const"],
        "tamper: mutated schema.const no longer exact-pins config",
    )
    check(
        file_grouped(schema_copy)
        != BOUND["schemas/governance/owner-decision-record-2026-08-16-v1.schema.json"],
        "tamper: mutated schema hash diverges from pin",
    )

    # 8h. Mutate manifest hash row.
    manifest_forged = json.loads(manifest_src.read_text("utf-8"))
    manifest_forged["artifacts"][0]["sha256"] = (
        "00000000:00000000:00000000:00000000:00000000:00000000:00000000:00000000"
    )
    manifest_copy = TAMPER_DIR / "mutated-manifest.json"
    manifest_copy.write_text(json.dumps(manifest_forged), "utf-8")
    observed_cfg = file_grouped(config_src)
    check(
        observed_cfg != manifest_forged["artifacts"][0]["sha256"],
        "tamper: forged manifest row no longer matches config bytes",
    )
    check(
        file_grouped(manifest_copy)
        != BOUND["configs/governance/owner-decision-record-2026-08-16-v1.hashes.json"],
        "tamper: mutated manifest hash diverges from pin",
    )

    # 8i. Invent an approval timestamp.
    forged_ts = json.loads(config_src.read_text("utf-8"))
    forged_ts["authority"]["approved_at"] = "2026-08-16T00:00:00Z"
    check(
        forged_ts["authority"]["approved_at"] is not None,
        "tamper setup: invented timestamp present in copy only",
    )
    check(
        json.loads(config_src.read_text("utf-8"))["authority"]["approved_at"] is None,
        "tamper: worktree approved_at remains null",
    )
    invented_copy = TAMPER_DIR / "invented-timestamp.json"
    invented_copy.write_text(json.dumps(forged_ts), "utf-8")
    check(
        file_grouped(invented_copy) != BOUND[config_src.relative_to(WORKTREE).as_posix()],
        "tamper: invented timestamp copy rejected by hash pin",
    )

    # Confirm worktree still clean of our copies (copies live under outputs/).
    check(TAMPER_DIR.is_relative_to(OUT_DIR), "tamper copies live only under outputs/A1")
    check(not TAMPER_DIR.is_relative_to(WORKTREE), "tamper copies are outside the worktree")

def verify_no_secrets() -> None:
    print("\n=== 10. Secret / credential / raw-data / broker-log scan of bound files ===")
    patterns = [
        re.compile(r"sk-[A-Za-z0-9]{16,}"),
        re.compile(r"-----BEGIN (?:RSA |OPENSSH |EC )?PRIVATE KEY-----"),
        re.compile(r"(?i)api[_-]?key\s*[:=]\s*['\"][A-Za-z0-9_\-]{12,}"),
        re.compile(r"(?i)secret\s*[:=]\s*['\"][^'\"]{8,}"),
        re.compile(r"(?i)password\s*[:=]\s*['\"][^'\"]+"),
        re.compile(r"AKIA[0-9A-Z]{16}"),
    ]
    hits = []
    for rel in BOUND:
        text = (WORKTREE / rel).read_text("utf-8", errors="replace")
        for pat in patterns:
            if pat.search(text):
                hits.append(f"{rel} matched {pat.pattern}")
    check(hits == [], "no secret/credential material in bound files", str(hits) if hits else "none")
    # D18 discusses key rotation but must not embed a key.
    d18 = (WORKTREE / "configs/governance/owner-decision-record-2026-08-16-v1.json").read_text()
    check("ROTATE_ALPHA_VANTAGE_KEY" in d18, "D18 discusses rotation (policy, not a secret)")
